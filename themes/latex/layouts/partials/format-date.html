{{- $date := .Date -}} {{- if and $date (not $date.IsZero) -}} {{- with site.Params.dateFormat -}} {{- $date.Format . -}} {{- else -}} {{- $day := $date.Day -}} {{- $suffix := "th" -}} {{- if in (slice 1 21 31) $day }}{{ $suffix = "st" }} {{- else if in (slice 2 22) $day }}{{ $suffix = "nd" }} {{- else if in (slice 3 23) $day }}{{ $suffix = "rd" }} {{- end -}} {{- printf "%d%s %s %d" $day $suffix ($date.Format "January") $date.Year -}} {{- end -}} {{- end -}}
